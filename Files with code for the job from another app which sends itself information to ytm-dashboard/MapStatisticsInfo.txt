package com.mallocinc.ytm.model.usecase.appInfo;

import com.mallocinc.ytm.database.generated.to.StatisticsLogTo;
import com.mallocinc.ytm.model.usecase.jobs.pojo.StatisticsLog;

import lombok.NoArgsConstructor;

@NoArgsConstructor
public class StatisticsLogMapper {

   public StatisticsLog buildReport(StatisticsLogTo to) {

      return StatisticsLog.builder().id(to.getID()).recordedAt(to.getRECORDED_AT()).userCnt(to.getUSER_CNT())
            .activeUsers(to.getACTIVE_USERS()).lockedUsers(to.getLOCKED_USERS()).suspendedUsers(to.getSUSPENDED_USERS())
            .pendingUsers(to.getPENDING_USERS()).inactiveUsers(to.getINACTIVE_USERS()).userGroupCnt(to.getUSER_GROUP_CNT())
            .rootUserGroupCnt(to.getROOT_USER_GROUP_CNT()).avgUsersPerGroup(to.getAVG_USERS_PER_GROUP())
            .adminCnt(to.getADMIN_CNT()).proctorCnt(to.getPROCTOR_CNT()).instructorCnt(to.getINSTRUCTOR_CNT())
            .studentCnt(to.getSTUDENT_CNT()).customRoleCnt(to.getCUSTOM_ROLE_CNT()).allPools(to.getALL_POOLS())
            .allRootPools(to.getALL_ROOT_POOLS()).allPoolsWithQuestions(to.getALL_POOLS_WITH_QUESTIONS())
            .totalQuestions(to.getTOTAL_QUESTIONS()).avgQuestionsInPools(to.getAVG_QUESTIONS_IN_POOLS())
            .singleChoiceQuestions(to.getSINGLE_CHOICE_QUESTIONS()).multipleChoiceQuestions(to.getMULTIPLE_CHOICE_QUESTIONS())
            .essayQuestions(to.getESSAY_QUESTIONS()).trueFalseQuestions(to.getTRUE_FALSE_QUESTIONS())
            .matrixQuestions(to.getMATRIX_QUESTIONS()).matchingQuestions(to.getMATCHING_QUESTIONS())
            .fillInBlanksQuestions(to.getFILL_IN_BLANKS_QUESTIONS()).orderingQuestions(to.getORDERING_QUESTIONS())
            .hotspotQuestions(to.getHOTSPOT_QUESTIONS()).openEndedQuestions(to.getOPEN_ENDED_QUESTIONS())
            .accountingQuestions(to.getACCOUNTING_QUESTIONS()).defCorrectincorrectScoring(to.getDEF_CORRECTINCORRECT_SCORING())
            .byAnswerWeightScoring(to.getBY_ANSWER_WEIGHT_SCORING()).manualScoring(to.getMANUAL_SCORING())
            .byCombinationScoring(to.getBY_COMBINATION_SCORING()).byRulesScoring(to.getBY_RULES_SCORING())
            .customQuestionProperties(to.getCUSTOM_QUESTION_PROPERTIES()).defaultTests(to.getDEFAULT_TESTS())
            .onlyTestsWithSections(to.getONLY_TESTS_WITH_SECTIONS()).allManualTests(to.getALL_MANUAL_TESTS())
            .allGeneratedTests(to.getALL_GENERATED_TESTS()).generatedByCategoryTests(to.getGENERATED_BY_CATEGORY_TESTS())
            .avgTestVersionsRootTests(to.getAVG_TEST_VERSIONS_ROOT_TESTS())
            .maxTestVersionsRootTests(to.getMAX_TEST_VERSIONS_ROOT_TESTS())
            .avgTestVersionsRootSec(to.getAVG_TEST_VERSIONS_ROOT_SEC()).maxTestVersionsRootSec(to.getMAX_TEST_VERSIONS_ROOT_SEC())
            .testTemplates(to.getTEST_TEMPLATES()).limitedTimeTests(to.getLIMITED_TIME_TESTS())
            .unlimitedTimeTests(to.getUNLIMITED_TIME_TESTS()).durationPerQuestionTests(to.getDURATION_PER_QUESTION_TESTS())
            .durationPerQuizTests(to.getDURATION_PER_QUIZ_TESTS()).deadlinePerQuizTests(to.getDEADLINE_PER_QUIZ_TESTS())
            .testOnePage(to.getTEST_ONE_PAGE()).testPagePerQuestion(to.getTEST_PAGE_PER_QUESTION())
            .afterTestReport(to.getAFTER_TEST_REPORT()).afterGradingReport(to.getAFTER_GRADING_REPORT())
            .afterVerificationReport(to.getAFTER_VERIFICATION_REPORT())
            .onManagerApprovalReport(to.getON_MANAGER_APPROVAL_REPORT()).realtimeGrading(to.getREALTIME_GRADING())
            .scoreDetails(to.getSCORE_DETAILS()).onlyScore(to.getONLY_SCORE()).testCertificates(to.getTEST_CERTIFICATES())
            .surveys(to.getSURVEYS()).selfEnrollmentGlobal(to.getSELF_ENROLLMENT_GLOBAL())
            .sessionUserGroups(to.getSESSION_USER_GROUPS()).sessionTraining(to.getSESSION_TRAINING())
            .proctoredu(to.getPROCTOREDU()).proctorio(to.getPROCTORIO()).favoriteReports(to.getFAVORITE_REPORTS())
            .scheduledReports(to.getSCHEDULED_REPORTS()).trainingCourses(to.getTRAINING_COURSES())
            .pinnedCourses(to.getPINNED_COURSES()).avgStepsPerTraining(to.getAVG_STEPS_PER_TRAINING())
            .avgTestsPerTraining(to.getAVG_TESTS_PER_TRAINING()).enabledEmailNotifications(to.getENABLED_EMAIL_NOTIFICATIONS())
            .disabledEmailNotifications(to.getDISABLED_EMAIL_NOTIFICATIONS())
            .enabledAppNotifications(to.getENABLED_APP_NOTIFICATIONS())
            .disabledAppNotifications(to.getDISABLED_APP_NOTIFICATIONS()).languageBundles(to.getLANGUAGE_BUNDLES())
            .usersLoggedAllTime(to.getUSERS_LOGGED_ALL_TIME()).usersLoggedLastMonth(to.getUSERS_LOGGED_LAST_MONTH())
            .usersCreatedLastMonth(to.getUSERS_CREATED_LAST_MONTH()).totalAttempts(to.getTOTAL_ATTEMPTS())
            .totalTestAttempts(to.getTOTAL_TEST_ATTEMPTS()).totalSurveyAttempts(to.getTOTAL_SURVEY_ATTEMPTS())
            .maxConcurrentUsersEver(to.getMAX_CONCURRENT_USERS_EVER())
            .maxConcurrentUsersLastMont(to.getMAX_CONCURRENT_USERS_LAST_MONT())
            .failedLoginsLastMonth(to.getFAILED_LOGINS_LAST_MONTH()).failedEmails(to.getFAILED_EMAILS())
            .unfinishedAttempts(to.getUNFINISHED_ATTEMPTS()).unfinishedDisconnectedAttemp(to.getUNFINISHED_DISCONNECTED_ATTEMP())
            .unfinishedInProgressAttemp(to.getUNFINISHED_IN_PROGRESS_ATTEMP())
            .unfinishedAvbContinueAttemp(to.getUNFINISHED_AVB_CONTINUE_ATTEMP())
            .unfinishedSuspendedAttemp(to.getUNFINISHED_SUSPENDED_ATTEMP()).hasStateDefined(to.getHAS_STATE_DEFINED())
            .hasCountryDefined(to.getHAS_COUNTRY_DEFINED()).build();
   }

}
