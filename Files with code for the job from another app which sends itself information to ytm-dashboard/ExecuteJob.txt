package com.mallocinc.ytm.model.usecase.jobs;

import java.math.BigDecimal;

import com.mallocinc.ytm.model.entities.job.Job;
import com.mallocinc.ytm.model.entities.job.JobInstance;
import com.mallocinc.ytm.model.providers.event.EventInstanceProvider;
import com.mallocinc.ytm.model.providers.jobs.JobInstanceProvider;
import com.mallocinc.ytm.model.providers.jobs.JobProvider;
import com.mallocinc.ytm.model.providers.jobs.JobProviderImpl;
import com.mallocinc.ytm.model.usecase.UseCase;
import com.mallocinc.ytm.model.usecase.jobs.ExecuteJob.Input;
import com.mallocinc.ytm.model.usecase.jobs.ExecuteJob.Output;

import lombok.AllArgsConstructor;
import lombok.Value;

public class ExecuteJob implements UseCase<Input, Output> {
   private final JobInstanceProvider   jobInstanceProvider = new JobInstanceProviderImpl();
   private final JobProvider           jobProvider         = new JobProviderImpl();
   private final EventInstanceProvider eventProvider       = new EventInstanceProvider();

   @AllArgsConstructor(staticName = "of")
   @Value
   public static class Input implements UseCase.InputValues {
      private final BigDecimal id;
   }

   @Value
   @AllArgsConstructor(staticName = "of")
   public static class Output implements UseCase.OutputValues {
      private final JobInstance jobInstance;
   }

   @Override
   public Output execute(Input input) {
      Job job = new GetJobById().execute(JobIdInput.of(input.getId())).getJob();
      JobExecutor jobExecutor = new JobExecutorFactory().create(job);
      JobInstance jobInstance = jobExecutor.execute();
      jobInstance = jobInstanceProvider.create(jobInstance);
      jobProvider.update(jobInstance.getJob());
      eventProvider.handleJobExecutedEvent(jobInstance);
      return Output.of(jobInstance);
   }

}
