   public List<VRepUserActionsTo> getActionInfoForPaging(int first,
         int pageSize,
         String sortField,
         String sortOrder,
         Map<String, Object> filters) throws MessageException {

      DataVRepUserActions dataUserActions = new DataVRepUserActions();
      Connection conn = null;
      List<VRepUserActionsTo> actions = new ArrayList<>();
      try {
         conn = Connect.obtainConn(conn);
         actions = dataUserActions.getActionInfoReportsForPaging(conn, first, pageSize, sortField, sortOrder, filters);
         Connect.returnConn(conn);
      } catch (GDAOException e) {

         throw new MessageException("error", MessageException.ERROR, "somethingWentWrong");
      } finally {
         Connect.returnConn(conn);
      }
      return actions;
   }

   public List<EntityUserActionLog> getUserActionLogInLogsWithoutLogOuts() throws MessageException {
      GdaoStatement userActionLogInLogsWithouyLogOuts = new GdaoStatement()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "LGN").and().isNot(UserActionLogTo.Columns.SESSION_ID.name(), null)
            .and().snippet(
                  new GdaoSnippetStatement("NOT EXISTS(SELECT * FROM USER_ACTION_LOG ual WHERE ual.session_id = user_action_log.session_id AND (ual.ACTION_CODE = 'LGO' OR ual.ACTION_CODE = 'POPUPTIME' OR ual.ACTION_CODE = 'POPUPCLICK' OR ual.ACTION_CODE = 'BRWSRSHUT' OR ual.ACTION_CODE = 'SYSCRASH' OR ual.ACTION_CODE = 'TCATSHUT'))"));
      return selectUsingGdaoStatement(userActionLogInLogsWithouyLogOuts);
   }

   public boolean doesSessionHaveLogOutRecord(String sessionID) throws MessageException {
      GdaoStatement sessionLogoutCountStatemet = new GdaoStatement().equal(UserActionLogTo.Columns.SESSION_ID.name(), sessionID)
            .and().leftParen().equal(UserActionLogTo.Columns.ACTION_CODE.name(), "POPUPCLICK").or()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "POPUPTIME").or()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "BRWSRSHUT").or()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "SYSCRASH").or()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "TCATSHUT").or()
            .equal(UserActionLogTo.Columns.ACTION_CODE.name(), "LGO").rightParen();
      long sessionLogoutCount = selectCountGdaoStatement(sessionLogoutCountStatemet);
      return sessionLogoutCount > 0;
   }

   public void createUserActionLogForSessionDestroy(String sessionID,
         BigDecimal userId,
         String ipAddress,
         EntityUser impersonator) {
      try {
         if (doesSessionHaveLogOutRecord(sessionID)) {
            logger.info("logout record exists for session: " + sessionID + ", sessiondestroy method will do nothing");
         } else {
            logUserActions(userId, UtilActionCodes.BROWSER_SHUT, UtilActionCodes.getActionText(UtilActionCodes.BROWSER_SHUT),
                  ipAddress, sessionID, checkImpersonatorId(impersonator), checkImpersonatorUsername(impersonator));
         }
      } catch (MessageException e) {
         throw new PersistenceException(e.getMessage(), e);
      }
   }

   public void createMissingLogoutsAtTomcatStop() throws MessageException {
      List<EntityUserActionLog> listOfUsers = getUserActionLogInLogsWithoutLogOuts();
      List<EntityUserActionLog> logoutRecords = listOfUsers.stream().map(this::createLogoutForLoginAtTomcatStop)
            .collect(Collectors.toList());
      batchEntityInsert(logoutRecords);
   }

            .collect(Collectors.toList());
      batchEntityInsert(logoutRecords);
   }

   public void attemptDisconnected(EntityQuizInstance leadQuizInstance) {
      String actionCode = leadQuizInstance.isSurvey() ? UtilActionCodes.SURVEY_LEFT : UtilActionCodes.TEST_LEFT;
      String actionDescription = UtilActionCodes.getActionText(actionCode + "." + "description",
            leadQuizInstance.getUniqueQuiz().getQuizDefinition().getQuizName());
      logUserActions(leadQuizInstance.getUserTestedId(), actionCode, actionDescription, null, leadQuizInstance.getId(), null,
            null, null);
   }

   public void attemptFinished(EntityQuizInstance leadQuizInstance, ApplicationUser applicationUser) {
      String actionCode = leadQuizInstance.isSurvey() ? UtilActionCodes.FINISH_SURVEY : UtilActionCodes.FINISH_TEST;
      String quizType = UtilActionCodes.getActionText(leadQuizInstance.isSurvey() ? "survey" : "test");
      logUserActions(applicationUser.getUserId(), actionCode,
            UtilActionCodes.getActionText(actionCode, quizType,
                  leadQuizInstance.getUniqueQuiz().getQuizDefinition().getQuizName()),
            applicationUser.getIpAddress(), leadQuizInstance.getId(), null,
            checkImpersonatorId(applicationUser.getImpersonator()), checkImpersonatorUsername(applicationUser.getImpersonator()));
   }

   private BigDecimal checkImpersonatorId(EntityUser impersonator) {
      return impersonator != null ? impersonator.getUserId() : null;
   }

   private String checkImpersonatorUsername(EntityUser impersonator) {
      return impersonator != null ? impersonator.getUsername() : null;
   }

   private void logAction(ApplicationUserData applicationUserModel,
         String actionCode,
         BigDecimal objectId,
         Object... parameters) {
      try {
         EntityUserActionLog actionLog = createLogObject(applicationUserModel, actionCode, objectId, parameters);
         insertEntity(actionLog);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   private EntityUserActionLog createLogObject(ApplicationUserData applicationUserModel,
         String actionCode,
         BigDecimal objectId,
         Object... parameters) {
      String text = UtilActionCodes.getActionText(actionCode, parameters);
      return getUserActionLogEntity(applicationUserModel.getUserId(), actionCode, text, applicationUserModel.getIpAddress(),
            objectId, applicationUserModel.getSessionId(), applicationUserModel.getBrowser(),
            applicationUserModel.getOperatingSystem(), checkImpersonatorId(applicationUserModel.getImpersonator()),
            checkImpersonatorUsername(applicationUserModel.getImpersonator()));

   }

   public void usersSettedToNotCompleted(ApplicationUser applicationUser, List<EntityVUserLesson> list) {
      try {

         Function<EntityVUserLesson, BigDecimal> getObjectIdFunction = x -> null;
         Function<EntityVUserLesson, Object[]> getParametersFunction = x -> new Object[] {
                                                                                           x.getTo().getUSERNAME(),
                                                                                           x.getLessonName() };

         createAndSaveActionLogs(applicationUser, UtilActionCodes.USERS_SETTED_TO_NOT_COMPLETED, list, getObjectIdFunction,
               getParametersFunction);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   private <T> void createAndSaveActionLogs(ApplicationUser applicationUser,
         String actionCode,
         List<T> list,
         Function<T, BigDecimal> getObjectIdFunction,
         Function<T, Object[]> getParametersFunction) throws MessageException {

      Function<T, EntityUserActionLog> createLogObjectFunction = x -> createLogObject(applicationUser, actionCode,
            getObjectIdFunction.apply(x), getParametersFunction.apply(x));
      List<EntityUserActionLog> actionLogs = list.stream().map(createLogObjectFunction).collect(Collectors.toList());
      batchEntityInsert(actionLogs);
   }

   public void userStatusSetted(ApplicationUser applicationUser, List<EntityVUserRole> list, String status) {
      try {
         List<EntityUserActionLog> logList = new ArrayList<>();
         for (EntityVUserRole user : list) {
            String text = UtilActionCodes.getActionText(UtilActionCodes.USER_STATUS_SETTED, user.getUser().getUsername(), status);
            EntityUserActionLog userActionLog = getUserActionLogEntity(applicationUser.getUserId(),
                  UtilActionCodes.USER_STATUS_SETTED, text, applicationUser.getIpAddress(), user.getUserId(),
                  applicationUser.getSessionId(), applicationUser.getBrowser(), applicationUser.getOperatingSystem(),
                  checkImpersonatorId(applicationUser.getImpersonator()),
                  checkImpersonatorUsername(applicationUser.getImpersonator()));
            logList.add(userActionLog);
         }
         batchEntityInsert(logList);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   /**
    * <p>Method logs each workspace members assignment to a workspace.
    * Logs are inserted in a batch.</p>
    * <p>This method only needs members id, username, workspace id and workspace name.
    * So, members object can be changed.</p>
    * 
    * @param    applicationUser - current user
    * @param    members - workspace members which membership has changed
    * @author   milos.ilic 
    * @since    11.1
    */
   public void logWorkspaceMembersCreated(ApplicationUser applicationUser, List<WorkspaceUserMemberRole> members) {
      try {
         List<EntityUserActionLog> actionLogs = new ArrayList<>();
         for (WorkspaceUserMemberRole member : members) {
            EntityUserActionLog userActionLog = new EntityUserActionLog(applicationUser.getUserId(),
                                                                        UtilActionCodes.WORKSPACE_MEMBER_ADDED,
                                                                        UtilActionCodes.getActionText(
                                                                              UtilActionCodes.WORKSPACE_MEMBER_ADDED,
                                                                              member.getUsername(), member.getWorkspaceName(),
                                                                              member.getWorkspaceRoleName()),
                                                                        applicationUser.getIpAddress());
            userActionLog.setSessionId(applicationUser.getSessionId());
            userActionLog.setObjectId(member.getWorkspaceId());
            actionLogs.add(userActionLog);
         }
         batchEntityInsert(actionLogs);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   public void usersAttemptsDeleted(ApplicationUser applicationUser, List<EntityVQuizInstanceResult> list) {
      try {
         List<EntityUserActionLog> logList = new ArrayList<>();
         for (EntityVQuizInstanceResult user : list) {
            EntityUserActionLog userActionLog = createLogObject(applicationUser, UtilActionCodes.ATTEMPT_DELETED,
                  user.getInstanceId(), user.getQuizName(), user.getQuizName());
            logList.add(userActionLog);
         }
         batchEntityInsert(logList);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }
