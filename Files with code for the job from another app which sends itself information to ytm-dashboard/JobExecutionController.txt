package com.mallocinc.ytm.webview.jobs;

import java.math.BigDecimal;
import java.sql.Timestamp;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.stream.Collectors;

import jakarta.faces.application.FacesMessage;

import com.mallocinc.ytm.model.entities.EntityVQuizInstanceResult;
import com.mallocinc.ytm.model.entities.job.Job;
import com.mallocinc.ytm.model.entities.job.JobInstance;
import com.mallocinc.ytm.model.providers.ActionLogProvider;
import com.mallocinc.ytm.model.providers.VQuizInstanceResultProvider;
import com.mallocinc.ytm.model.usecase.ListInput;
import com.mallocinc.ytm.model.usecase.jobs.ExecuteJob;
import com.mallocinc.ytm.model.usecase.jobs.JobType;
import com.mallocinc.ytm.model.usecase.jobs.ListJobs;
import com.mallocinc.ytm.webview.controllers.setup.Controller;
import com.mallocinc.ytm.webview.controllers.setup.FacesUtil;

import lombok.Getter;
import lombok.Setter;

public class JobExecutionController extends Controller {

   private static final long serialVersionUID = 1L;

   public JobExecutionController() {
      Map<String, Object> jobsFilter = createJobFiltersMap();
      jobList = new ListJobs().execute(ListInput.builder().listAll(true).filterBy(jobsFilter).build()).getList();
      jobNamesMap = jobList.stream().collect(Collectors.toMap(j -> j.getName() + ": " + j.getId().intValue(), Job::getId));
   }

   @Getter
   @Setter
   private BigDecimal                    selectedJobId;
   @Getter
   private Job                           selectedJob;

   @Getter
   private final Map<String, BigDecimal> jobNamesMap;

   private final List<Job>               jobList;

   @Getter
   private JobInstanceDTO                executedJobInstance;

   private int                           numberOfRecords     = 0;

   @Getter
   private String                        confirmationMessage = "system.jobs.actions.execute.confirm";

   public void executeJob() {
      JobInstance instance = new ExecuteJob().execute(ExecuteJob.Input.of(selectedJobId)).getJobInstance();
      executedJobInstance = new JobInstanceDTO(instance);

      if (instance.getErrorCount() == 0) {
         FacesUtil.throwMessageFormatted("info", FacesMessage.SEVERITY_INFO,
               "eTestingCenter.gradeCenter.jobExecution.execute.success", numberOfRecords);
         logTestResultsExportedViaSystemJob(instance.getJob());
      } else {
         FacesUtil.throwWarning("system.jobs.actions.execute.errors");
      }
   }

   private void logTestResultsExportedViaSystemJob(Job job) {
      new ActionLogProvider().testResultsExportedViaSystemJob(getApplicationUser(), job);
   }

   public boolean isJobSelected() {
      return selectedJobId != null && !selectedJobId.equals(BigDecimal.ZERO);
   }

   private Map<String, Object> createJobFiltersMap() {
      Map<String, Object> jobsFilter = new HashMap<>();

      jobsFilter.put("type", JobType.EXPORT_TEST_RESULTS_WITH_POOL_BREAKDOWN.name());

      return jobsFilter;
   }

   public boolean isJobExecuted() {
      return executedJobInstance != null;
   }

   /**
    * Handles the change of the selected job, updating relevant information such as the
    * number of verified attempts after the last job execution and generating a confirmation message.
    *
    * If a selected job is found, it retrieves details such as the last execution timestamp and
    * a flag indicating whether only modified records should be considered. It then queries and
    * counts the number of authorized and verified attempts updated after the last modification.
    * The result is stored in the {@code numberOfRecords} field, and a confirmation message is
    * generated using the provided resource key.
    * 
    */
   public void handleJobChange() {
      if (isJobSelected()) {
         Optional<Timestamp> lastModified = Optional.empty();
         selectedJob = jobList.stream().filter(j -> j.getId().equals(selectedJobId)).findFirst().get();
         Boolean modifiedOnly = (Boolean) selectedJob.getData().get("modifiedOnly");
         if (selectedJob.getLastExecuted() != null && modifiedOnly) {
            lastModified = Optional.of(selectedJob.getLastExecuted());
         }
         List<EntityVQuizInstanceResult> records = new VQuizInstanceResultProvider()
               .listAllAuthorizedAndVerifiedAttemptsUpdatedAfter(lastModified);
         numberOfRecords = records.size();
         confirmationMessage = FacesUtil.getMessage("eTestingCenter.gradeCenter.jobExecution.execute.confirmation",
               numberOfRecords);
      }
   }

}
