package com.mallocinc.ytm.model.usecase.jobs;

import java.io.IOException;
import java.net.SocketTimeoutException;
import java.net.UnknownHostException;
import java.nio.charset.StandardCharsets;
import java.util.UUID;

import javax.net.ssl.SSLHandshakeException;

import org.apache.http.HttpEntity;
import org.apache.http.client.config.RequestConfig;
import org.apache.http.client.methods.CloseableHttpResponse;
import org.apache.http.client.methods.HttpPost;
import org.apache.http.conn.ConnectTimeoutException;
import org.apache.http.entity.ContentType;
import org.apache.http.entity.StringEntity;
import org.apache.http.impl.client.CloseableHttpClient;
import org.apache.http.impl.client.HttpClients;
import org.apache.http.util.EntityUtils;

import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.mallocinc.ytm.database.generated.to.StatisticsLogTo;
import com.mallocinc.ytm.model.entities.job.Job;
import com.mallocinc.ytm.model.providers.StatisticsLogProvider;
import com.mallocinc.ytm.model.usecase.appInfo.StatisticsLogMapper;
import com.mallocinc.ytm.model.usecase.jobs.pojo.StatisticsLog;
import com.mallocinc.ytm.model.util.ModelException;

import lombok.extern.log4j.Log4j2;

@Log4j2
public class SendStatisticsInfoJobExecutor extends JobExecutor {

   private final StatisticsLogMapper   statsMapper        = new StatisticsLogMapper();
   private final String                endpointUrl;
   private final StatisticsLogProvider statisticsProvider = new StatisticsLogProvider();

   protected SendStatisticsInfoJobExecutor(Job job) {
      super(job);

      this.endpointUrl = getJob().getData().get("endpointUrl").toString();
   }

   @Override
   protected void executeImpl() {
      try {

         StatisticsLogTo to = statisticsProvider.createAndFetchCurrent();

         StatisticsLog report = statsMapper.buildReport(to);

         postStatisticsInfo(report);

      } catch (Exception e) {
         log.error(e.getMessage(), e);
         throw new ModelException(e);
      }
   }

   private static String abbreviate(String s, int max) {
      if (s == null) {
         return null;
      }
      return s.length() > max ? s.substring(0, max) : s;
   }

   private void postStatisticsInfo(StatisticsLog statisticsReport) {
      log.info("Post Statistics Info - Started");
      RequestConfig config = RequestConfig.custom().setConnectTimeout(8_000).setConnectionRequestTimeout(8_000)
            .setSocketTimeout(12_000).build();

      try (CloseableHttpClient httpClient = HttpClients.custom().setDefaultRequestConfig(config).build()) {
         HttpPost httpPost = new HttpPost(this.endpointUrl);
         httpPost.setConfig(config);
         httpPost.setHeader("Content-Type", "application/json");
         httpPost.setHeader("Accept", "application/json");
         httpPost.setHeader("User-Agent", "ytm-app/1.0");
         httpPost.setHeader("X-Request-Id", UUID.randomUUID().toString());
         //httpPost.setHeader("Authorization", "Bearer " + accessToken.getToken());

         ObjectMapper objectMapper = new ObjectMapper();
         objectMapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);
         String jsonInputString = objectMapper.writeValueAsString(statisticsReport);
         HttpEntity entity = new StringEntity(jsonInputString, ContentType.APPLICATION_JSON.withCharset(StandardCharsets.UTF_8));
         httpPost.setEntity(entity);

         httpPost.setHeader("Content-Type", "application/json; charset=UTF-8");

         try (CloseableHttpResponse response = httpClient.execute(httpPost)) {
            int statusCode = response.getStatusLine().getStatusCode();
            String responseBody = EntityUtils.toString(response.getEntity(), StandardCharsets.UTF_8);
            if (statusCode / 100 != 2) {
               throw new ModelException(String.format("HTTP %d from dashboard at %s. Body: %s", statusCode, this.endpointUrl,
                     abbreviate(responseBody, 300)));
            }
            log.info("Post Statistics Info - Finished");

         }
      } catch (UnknownHostException e) {
         throw new ModelException("DNSResolutionError: " + e.getMessage(), e);
      } catch (ConnectTimeoutException | SocketTimeoutException e) {
         throw new ModelException("TimeoutError: " + e.getMessage(), e);
      } catch (SSLHandshakeException e) {
         throw new ModelException("TLSCertificateError: " + e.getMessage(), e);
      } catch (IOException e) {
         throw new ModelException("NetworkError: " + e.getMessage(), e);
      }

   }

}
