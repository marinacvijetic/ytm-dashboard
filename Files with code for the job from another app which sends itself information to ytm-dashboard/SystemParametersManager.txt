/**
 * Global singleton responsible for management of system parameters. All system
 * parameters are loaded from database and stored in the memory. On every
 * parameter value change, parameter will be updated in the database.
 * 
 * @author ytm
 *
 */
public class SystemParametersManager {

   private static final int                   DEFAULT_SESSION_TIMEOUT_MINUTES = 180;
   private static final Logger                logger                          = Logger.getLogger(SystemParametersManager.class);
   private static SystemParametersManager     instance;
   @Getter

   /**
    * Map of system parameters, key is code of system parameter
    */
   private Map<String, EntitySystemParameter> systemParametersMap;

   private final WebhookCallsProvider         webhookCallsProvider            = new WebhookCallsProvider();

   /**
    * Private constructor to forbid instantiation outside the class
    */
   private SystemParametersManager() {
      loadSystemParameterMap();

   }

   /**
    * Loads system parameters map from database.
    */
   private void loadSystemParameterMap() {
      synchronized (this) {
         systemParametersMap = new SystemParameterProvider().getSystemParametersMap();
      }
   }

   /**
    * Reloads system parameters from the database in the local service cache. If the system parameters are changed directly in the database 
    * or other service in the Tomcat cluster, with this method system parameters are reloaded in the service.
    */
   public void reloadParameters() {
      loadSystemParameterMap();
   }

   /**
    * Reloads system parameters from the database in the local service cahce for this service and all other services in the tomcat cluster via webhooks.
    */
   public void reloadParametersSystemWide() {
      reloadParameters();
      webhookCallsProvider.handleSystemParameterUpdate();
   }

   /**
    * Return instance of SystemParameterManager. Creates instance if is not created
    * yet.
    * 
    * @return instance of SystemParameterManager
    */
   public static synchronized SystemParametersManager getInstance() {
      if (instance == null) {
         instance = new SystemParametersManager();
      }
      return instance;
   }

   /**
    * Returns String value of parameter specified by code.
    * 
    * @param code system parameter code
    * @return parameter value if exists, otherwise null
    */
   public String getStringParameterValue(String code) {
      return applyFunction(EntitySystemParameter::getValue, code);
   }

   /**
    * Returns BigDecimal value of parameter specified by code.
    * 
    * @param code system parameter code
    * @return parameter value if exists, otherwise null
    */
   public BigDecimal getNumericParameterValue(String code) {
      return applyFunction(EntitySystemParameter::getNumericValue, code);
   }

   /**
    * Returns blob value of parameter specified by code.
    * 
    * @param code system parameter code
    * @return parameter value if exists, otherwise null
    */
   public byte[] getBlobParameterValue(String code) {
      return applyFunction(EntitySystemParameter::getBlobValue, code);
   }

   /**
    * Returns boolean value of parameter specified by code.
    * 
    * @param code system parameter code
    * @return parameter value if exists, otherwise null
    */
   public boolean getBooleanParameterValue(String code) {
      return applyFunction(EntitySystemParameter::getBooleanValue, code);
   }

   /**
    * Applies provided function on system parameter specified by code.
    * 
    * @param <T>      Type of result of the function.
    * @param function function to apply
    * @param code     system parameter code
    * @return result of applied function
    */
   private <T> T applyFunction(Function<EntitySystemParameter, T> function, String code) {
      Optional<EntitySystemParameter> systemParameterOpt = getByCode(code);
      return systemParameterOpt.isPresent() ? function.apply(systemParameterOpt.get()) : null;
   }

   /**
    * Changes String value of system parameter specified by code. Change will be
    * propagated to database.
    * 
    * @param code  system parameter code
    * @param value new value of system parameter
    * @throws MessageException
    */
   public void updateParameterValue(String code, String value) {
      updateParameterValue(parameter -> parameter.setValue(value), code);
   }

   /**
    * Changes BigDecimal value of system parameter specified by code. Change will
    * be propagated to database.
    * 
    * @param code  system parameter code
    * @param value new value of system parameter
    * @throws MessageException
    */
   public void updateParameterValue(String code, BigDecimal value) {
      updateParameterValue(parameter -> parameter.setNumericValue(value), code);
   }

   /**
    * Changes blob value of system parameter specified by code. Change will be
    * propagated to database.
    * 
    * @param code  system parameter code
    * @param value new value of system parameter
    * @throws MessageException
    */
   public void updateParameterValue(String code, byte[] value) {
      updateParameterValue(parameter -> parameter.setBlobValue(value), code);
   }

   /**
    * Changes Boolean value of system parameter specified by code. Change will be
    * propagated to database.
    * 
    * @param code  system parameter code
    * @param value new value of system parameter
    * @throws MessageException
    */
   public void updateParameterValue(String code, boolean value) {
      updateParameterValue(parameter -> parameter.setBooleanValue(value), code);
   }

   /**
    * Changes value of system parameter specified by code. Change will be
    * propagated to database. Type of value to change is specified with
    * valueChanger consumer.
    * 
    * @param valueChanger consumer which changes the value of system parameter
    * @param code
    * @throws MessageException
    */
   private void updateParameterValue(Consumer<EntitySystemParameter> valueChanger, String code) {
      try {
         synchronized (this) {
            Optional<EntitySystemParameter> systemParameterOpt = getByCode(code);
            if (systemParameterOpt.isPresent()) {
               EntitySystemParameter parameter = systemParameterOpt.get();
               valueChanger.accept(parameter);
               parameter.update(null);
               parameter.setDirty(false);
               webhookCallsProvider.handleSystemParameterUpdate();
               logger.debug(String.format("System parameter %s updated in database.", parameter.getCode()));
            } else {
               logger.debug(String.format("System parameter %s not found.", code));
            }
         }
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   /**
    * Returns system parameter entity from system parameter map
    * 
    * @param code system parameter code
    * @return
    */
   public Optional<EntitySystemParameter> getByCode(String code) {
      return Optional.ofNullable(systemParametersMap.get(code));
   }

   public void setParameterValue(String code, String value) {
      setParameterValue(parameter -> parameter.setValue(value), code);
   }

   public void setParameterValue(String code, BigDecimal value) {
      setParameterValue(parameter -> parameter.setNumericValue(value), code);
   }

   public void setParameterValue(String code, byte[] value) {
      setParameterValue(parameter -> parameter.setBlobValue(value), code);
   }

   public void setParameterValue(String code, boolean value) {
      setParameterValue(parameter -> parameter.setBooleanValue(value), code);
   }

   private void setParameterValue(Consumer<EntitySystemParameter> valueChanger, String code) {
      synchronized (this) {
         Optional<EntitySystemParameter> systemParameterOpt = getByCode(code);
         if (systemParameterOpt.isPresent()) {
            EntitySystemParameter parameter = systemParameterOpt.get();
            valueChanger.accept(parameter);
            parameter.setDirty(true);
            logger.debug("System parameter " + parameter.getCode() + " changed in SystemParametersMap.");
         } else {
            logger.debug("System parameter " + code + " not found.");
         }
      }
   }

   public void saveDirtyParameters() {
      try {
         synchronized (this) {
            List<EntitySystemParameter> dirtyParameters = systemParametersMap.values().stream()
                  .filter(EntitySystemParameter::isDirty).collect(Collectors.toList());
            new SystemParameterProvider().batchEntityUpdate(dirtyParameters);
            dirtyParameters.forEach(entity -> entity.setDirty(false));
            webhookCallsProvider.handleSystemParameterUpdate();
         }
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

      setParameterValue(UtilSystemParameter.PROCTORING_TEMPLATE, configuration.getTemplate());
      setParameterValue(UtilSystemParameter.PROCTORING_LIVE_PROCTORS_EMAIL_REGEX, configuration.getLiveProctorsEmailRegex());

      setParameterValue(UtilSystemParameter.PROCTORING_FULL_NAME_AS_NICKNAME, configuration.isFullNameAsNickname());
      saveDirtyParameters();
   }

   public static void updatePasswordConfiguration(PasswordConfiguration configuration) {
      if (configuration.getMinimalPasswordLenght() <= MAX_PASSWORD_LENGHT
          && configuration.sumOfMinimalNumberChars() <= MAX_PASSWORD_LENGHT) {
         getInstance().setParameterValue(MINIMAL_PASSWORD_LENGHT, BigDecimal.valueOf(configuration.getMinimalPasswordLenght()));
         getInstance().setParameterValue(MINIMAL_NUMBER_PASSWORD_UPPERCASE_CHARACTERS,
               BigDecimal.valueOf(configuration.getMinimalNumberPasswordUppercaseCharacters()));
         getInstance().setParameterValue(MINIMAL_NUMBER_PASSWORD_LOWERCASE_CHARACTERS,
               BigDecimal.valueOf(configuration.getMinimalNumberPasswordLowercaseCharacters()));
         getInstance().setParameterValue(MINIMAL_NUMBER_PASSWORD_DIGIT_CHARACTERS,
               BigDecimal.valueOf(configuration.getMinimalNumberPasswordDigitCharacters()));
         getInstance().setParameterValue(MINIMAL_NUMBER_PASSWORD_SPECIAL_CHARACTERS,
               BigDecimal.valueOf(configuration.getMinimalNumberPasswordSpecialCharacters()));
         getInstance().saveDirtyParameters();
      } else {
         throw new ModelException("The sum of the minimum specified number of characters for each character type is larger than the maximal password length, or the minimal password length is larger than the maximal password length.");
      }
   }

   public static GreenhouseIntegrationParameters updateGreenhouseIntegrationParameters(
         GreenhouseIntegrationParameters parameters) {
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_ENABLED, parameters.isEnabled());
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_API_KEY, parameters.getApiKey());
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_WEBHOOK_ENABLED, parameters.isWebhookEnabled());
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_WEBHOOK_USERNAME, parameters.getWebhookUsername());
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_WEBHOOK_PASSWORD, parameters.getWebhookPassword());
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_TEST_DELIVERY_PREFERENCE,
            Optional.ofNullable(parameters.getDeliveryPreference()).map(QuizDeliveryPreference::name).orElse(null));
      getInstance().setParameterValue(GREENHOUSE_INTEGRATION_GRADE_ATTEMPT_EXPIRATION,
            Optional.ofNullable(parameters.getGradeAttemptExpirationTime()).map(Timestamp::toString).orElse(null));
      getInstance().saveDirtyParameters();
      EventDefinition eventDefinition = EventDefinitionManager.getGreenhouseIntegrationErrorEvent();
      eventDefinition.setNotificationEnabled(parameters.isEnabled());
      EventDefinitionManager.getInstance().updateEventDefinition(eventDefinition);
      return getGreenhouseIntegrationParameters();
   }

   /**
    * Checks if the RCDSO integration module is enabled and configured and throws exception if not. 
    * All parameters are required so it is sufficient to check if any is null.
    */
   public static void validateRcdsoModule() {
      if (!isRcdsoIntegrationEnabled()) {
         throw new ModelException("The RCDSO integration module is not enabled.");
      }
      if (getRcdsoAppId() == null) {
         throw new ModelException("The RCDSO integration module is not configured. The integration parameters should be configured under System -> Integrations -> RCDSO integration settings.");
      }
   }

   public static boolean isRcdsoIntegrationEnabledAndConfigured() {
