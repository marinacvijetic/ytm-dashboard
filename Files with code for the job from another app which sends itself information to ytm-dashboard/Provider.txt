   protected <D extends DataYtmSample<T>, T extends BaseToClass, E extends Entity> List<E> selectUsingGdaoStatement(
         GdaoStatement gdaoStatement,
         Connection connection,
         Class<D> dataClass,
         Function<T, E> mapper) throws MessageException {
      try {
         logger.debug("Select query: " + gdaoStatement.getQuery());
         logger.debug("Select prepared query: " + gdaoStatement.getPreparedStatementQuery());
         D data = getInstanceOfData(dataClass);

         List<T> resultTo = data.selectUsingQuery(connection, gdaoStatement.getPreparedStatementQuery(),
               gdaoStatement.getParameters());

         return resultTo.stream().map(mapper).collect(Collectors.toList());
      } catch (GDAOException e) {
         logger.error(e.getMessage(), e);
         throw new MessageException(SOMETHING_WENT_WRONG);
      }
   }

      } catch (GDAOException e) {
         logger.error(e.getMessage(), e);
         throw new MessageException(SOMETHING_WENT_WRONG);
      }
   }

         throw new RuntimeException(e);

   protected <D extends DataYtmSample<T>, T extends BaseToClass, E extends Entity> ListOutput<E> list(ListInput input,
         Map<String, String> columnsMapping,
         Class<D> dataClass,
         Function<T, E> mapper) {
      try {
         GdaoStatement filterStatement = createFilterStatement(input, columnsMapping);
         GdaoPagingStatement listStatement = new GdaoPagingStatement(input.getFirst(), input.getPageSize());
         if (input.getSortBy() != null && !input.getSortBy().isEmpty()) {
            Optional<Entry<String, javax.swing.SortOrder>> sort = input.getSortBy().entrySet().stream().findFirst();
            String sortField = sort.isPresent() ? columnsMapping.get(sort.get().getKey()) : null;
            SortOrder sortOrder = sort.isPresent() ? sort.get().getValue() : null;
            listStatement.setOrder(sortField, convertSortOrder(sortOrder));
         }
         listStatement.and(filterStatement);
         List<E> list = selectUsingGdaoStatement(listStatement, dataClass, mapper);
         //This line is not needed since the count is done before the data load - review required by @adis
         //long total = selectCountGdaoStatement(filterStatement, dataClass);
         long total = list.size();

         return new ListOutput<>(list, total);
      } catch (MessageException e) {
         throw new PersistenceException(e);
      }
   }

   protected GdaoStatement createFilterStatement(ListInput input, Map<String, String> columnsMapping) {
      if (input.getFilterBy() == null) {
         return new GdaoStatement();
      }
      Set<Entry<String, Object>> entrySet = input.getFilterBy().entrySet();
      List<GdaoStatement> filterStatements = entrySet.stream().map(entry -> createTerm(entry, columnsMapping))
            .collect(Collectors.toList());
      return filterStatements.stream().reduce(new GdaoStatement(), GdaoStatement::and);
   }

   protected OrderCriteria convertSortOrder(SortOrder sortOrder) {
      return (sortOrder != null && sortOrder == SortOrder.DESCENDING) ? OrderCriteria.DESCENDING : OrderCriteria.ASCENDING;
   }

   protected GdaoStatement createTerm(Map.Entry<String, Object> entry, Map<String, String> columnsMapping) {
      final String dbColumnName = Optional.ofNullable(columnsMapping.get(entry.getKey())).orElse(entry.getKey());

      if (entry.getValue() instanceof NullFilter) {
         return new GdaoStatement().is(columnsMapping.get(entry.getKey()), null);
      }
      if (entry.getValue() instanceof Boolean) {
         return new GdaoStatement().equal(dbColumnName, entry.getValue());
      }
      if (entry.getValue() instanceof BigDecimal) {
         return new GdaoStatement().equal(dbColumnName, entry.getValue());
      }
      if (entry.getValue() instanceof String && BooleanUtils.toBooleanObject((String) entry.getValue()) != null) {
         return new GdaoStatement().equal(dbColumnName, BooleanUtils.toBooleanObject((String) entry.getValue()));
      }
      if (entry.getValue() instanceof NotEqualFilter) {
         return new GdaoStatement().notEqual(dbColumnName, ((NotEqualFilter) entry.getValue()).getFilterValue());
      }
      if (entry.getValue() instanceof EqualFilter) {
         return new GdaoStatement().equal(dbColumnName, ((EqualFilter) entry.getValue()).getFilterValue());
      }
      if (entry.getValue() instanceof InFilter) {
         List<Object> filterValues = ((InFilter) entry.getValue()).getFilterValues();
         if (filterValues.isEmpty()) {
            return new GdaoStatement().snippet(new GdaoSnippetStatement("FALSE"));
         } else {
            return new GdaoStatement().leftParen()
                  .concat(filterValues.stream()
                        .map(value -> value == null
                                                    ? new GdaoStatement().is(dbColumnName, value)
                                                    : new GdaoStatement().equal(dbColumnName, value))
                        .reduce(new GdaoStatement(), GdaoStatement::or))
                  .rightParen();
         }
      }
      if (entry.getValue() instanceof GreaterOrEqualFilter) {
         return new GdaoStatement().greaterOrEqual(dbColumnName, ((GreaterOrEqualFilter) entry.getValue()).getFilterValue());
      }
      if (entry.getValue() instanceof LessOrEqualFilter) {
         return new GdaoStatement().lessOrEqual(dbColumnName, ((LessOrEqualFilter) entry.getValue()).getFilterValue());
      }

      return new GdaoStatement().like(dbColumnName, entry.getValue());
   }
